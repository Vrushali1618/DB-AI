import math

# Function to print the board
def print_board(board):
    for row in [board[i:i+3] for i in range(0, 9, 3)]:
        print("|".join(row))
    print()

# Check for winner
def check_winner(board):
    win_states = [
        [0, 1, 2], [3, 4, 5], [6, 7, 8],  # Rows
        [0, 3, 6], [1, 4, 7], [2, 5, 8],  # Columns
        [0, 4, 8], [2, 4, 6]              # Diagonals
    ]

    for state in win_states:
        if board[state[0]] == board[state[1]] == board[state[2]] != " ":
            return board[state[0]]

    if " " not in board:
        return "Tie"

    return None

# Alpha-Beta Minimax Algorithm
def alpha_beta(board, depth, alpha, beta, maximizing_player):

    winner = check_winner(board)

    if winner == "X":
        return 1
    elif winner == "O":
        return -1
    elif winner == "Tie":
        return 0

    # Maximizing Player (AI)
    if maximizing_player:
        max_eval = -math.inf

        for i in range(9):
            if board[i] == " ":
                board[i] = "X"

                eval = alpha_beta(board, depth + 1, alpha, beta, False)

                board[i] = " "

                max_eval = max(max_eval, eval)
                alpha = max(alpha, eval)

                # Pruning
                if beta <= alpha:
                    break

        return max_eval

    # Minimizing Player (Human)
    else:
        min_eval = math.inf

        for i in range(9):
            if board[i] == " ":
                board[i] = "O"

                eval = alpha_beta(board, depth + 1, alpha, beta, True)

                board[i] = " "

                min_eval = min(min_eval, eval)
                beta = min(beta, eval)

                # Pruning
                if beta <= alpha:
                    break

        return min_eval

# Find best move for AI (X)
def best_move(board):

    best_val = -math.inf
    move = -1

    for i in range(9):

        if board[i] == " ":

            board[i] = "X"

            move_val = alpha_beta(board, 0, -math.inf, math.inf, False)

            board[i] = " "

            if move_val > best_val:
                best_val = move_val
                move = i

    return move

# Game loop
def play_game():

    board = [" "] * 9

    human = "O"
    ai = "X"

    print("Tic-Tac-Toe! You are O, AI is X")
    print("Board positions are:")
    print("0 | 1 | 2")
    print("3 | 4 | 5")
    print("6 | 7 | 8\n")

    print_board(board)

    while True:

        # Human move
        move = int(input("Enter your move (0-8): "))

        if move < 0 or move > 8 or board[move] != " ":
            print("Invalid move. Try again.")
            continue

        board[move] = human

        print_board(board)

        result = check_winner(board)

        if result:
            print("Winner:", result)
            break

        # AI move
        ai_move = best_move(board)

        board[ai_move] = ai

        print("AI plays:")

        print_board(board)

        result = check_winner(board)

        if result:
            print("Winner:", result)
            break

# Driver Code
if __name__ == "__main__":
    play_game()

# alpha -> maximizing player
# beta  -> minimizing player
